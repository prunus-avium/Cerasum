cmake_minimum_required(VERSION 3.30)
project(Cerasum VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

find_program(CLANG_TIDY_EXE NAMES clang-tidy)

if(CLANG_TIDY_EXE)
    message(STATUS "Found clang-tidy: ${CLANG_TIDY_EXE}")
    set(CMAKE_CXX_CLANG_TIDY "${CLANG_TIDY_EXE}")
endif()

include(FetchContent)
include(CTest)

find_package(glfw3 QUIET)
if (glfw3_FOUND)
    message(STATUS "Found glfw3 successfully.")
else()
    message(STATUS "glfw3 not found. Downloading and building it automatically.")
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4)
    FetchContent_MakeAvailable(glfw)
endif()
find_package(OpenAL CONFIG QUIET)
if (OPENAL_FOUND)
    message(STATUS "Found OpenAL successfully.")
else()
    message(STATUS "OpenAL not found. Downloading and building it automatically.")
    FetchContent_Declare(openal GIT_REPOSITORY https://github.com/kcat/openal-soft.git GIT_TAG 1.25.2)
    FetchContent_MakeAvailable(openal)
endif()
find_package(Vulkan REQUIRED)
if (NOT Vulkan_FOUND)
    message(FATAL_ERROR "Cannot automatically download Vulkan SDK. Please install it manually.")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG QUIET
    libavcodec
    libavformat
    libavutil
    libswresample
)
if (FFMPEG_FOUND)
    message(STATUS "Found FFMPEG successfully.")
    add_library(ffmpeg_interface INTERFACE)
    target_include_directories(ffmpeg_interface INTERFACE ${FFMPEG_INCLUDE_DIRS})
    target_link_directories(ffmpeg_interface INTERFACE ${FFMPEG_LIBRARY_DIRS})
    target_link_libraries(ffmpeg_interface INTERFACE ${FFMPEG_LIBRARIES})
    target_compile_options(ffmpeg_interface INTERFACE ${FFMPEG_CFLAGS_OTHER})
else()
    message(FATAL_ERROR "Cannot automatically download FFMPEG. Please install it manually.")
endif()
find_package(fmt QUIET)
if (fmt_FOUND)
    message(STATUS "Found fmt successfully.")
else()
    message(STATUS "fmt not found. Downloading and building it automatically.")
    FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG 12.2.0)
    FetchContent_MakeAvailable(fmt)
endif()
find_package(spdlog QUIET)
if (spdlog_FOUND)
    message(STATUS "Found spdlog successfully.")
else()
    message(STATUS "spdlog not found. Downloading and building it automatically.")
    FetchContent_Declare(spdlog GIT_REPOSITORY https://github.com/gabime/spdlog.git GIT_TAG v1.17.0)
    FetchContent_MakeAvailable(spdlog)
endif()
find_package(nlohmann_json QUIET)
if (nlohmann_json_FOUND)
    message(STATUS "Found nlohmann_json successfully.")
else()
    message(STATUS "nlohmann_json not found. Downloading and building it automatically.")
    FetchContent_Declare(nlohmann_json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0)
    FetchContent_MakeAvailable(nlohmann_json)
endif()

add_subdirectory(utils)
add_subdirectory(core)
add_subdirectory(decoder)
add_subdirectory(player)
add_subdirectory(ui)
add_subdirectory(app)

if (BUILD_TESTING)
    find_package(GTest QUIET)
    if (GTest_FOUND)
        message(STATUS "Found GTest successfully.")
    else()
        message(STATUS "GTest not found. Downloading and building it automatically.")
        FetchContent_Declare(googletest GIT_REPOSITORY https://github.com/google/googletest.git GIT_TAG v1.17.0)
        FetchContent_MakeAvailable(googletest)
    endif()
    enable_testing()
    add_subdirectory(test)
endif()
